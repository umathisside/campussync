return newAlert; } } return null; } // POST /api/attendance - Batch record class attendance app.post('/api/attendance', (req, res) => { try { const { date, lectureId, records } = req.body; if (!date || !lectureId || !Array.isArray(records)) { return res.status(400).json({ error: 'Missing date, lectureId, or records array' });
